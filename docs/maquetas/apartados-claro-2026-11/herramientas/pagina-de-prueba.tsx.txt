const item = (n: number, cantidad = 1, descuento = 0, extra: Partial<PrendaApartada> = {}): PrendaApartada => {
  const p = PRENDAS[n - 1];
  return { itemId: `i-${n}-${Math.round(descuento * 100)}`, varianteId: p.varianteId, sku: p.sku, referencia: p.referencia, cantidad, precioUnitario: p.precio, descuentoUnitario: descuento, motivoDescuento: null, motivoDescuentoDetalle: null, argumentoDescuento: null, descuentoGrupoUnitario: 0, ofertaGrupo: null, ...extra };
};

const apartado = (n: number, nombres: string, apellidos: string, estado: Apartado["estado"], venceEl: string, creadaEn: string, prendas: PrendaApartada[], adelanto: number, extra: Partial<Apartado> = {}): Apartado => {
  const total = Math.round(prendas.reduce((s, p) => s + (p.precioUnitario - p.descuentoUnitario) * p.cantidad, 0) * 100) / 100;
  return {
    id: `a-${n}`,
    codigo: `APT-TRU-${String(n).padStart(4, "0")}`,
    estado,
    nombres,
    apellidos,
    celular: `9${String(87000000 + n * 1111).padStart(8, "0")}`,
    dni: n % 3 === 0 ? null : String(40000000 + n * 12345),
    asesora: n % 2 ? "Micaela" : "Rosa",
    total,
    adelanto,
    saldo: Math.round((total - adelanto) * 100) / 100,
    venceEl,
    extensiones: 0,
    creadaEn: `${creadaEn}T16:20:00Z`,
    devolucionMedio: "yape",
    devolucionNumero: `9${String(87000000 + n * 1111).padStart(8, "0")}`,
    devolucionCciFinal: null,
    liberadaSola: false,
    comprobanteAnticipo: null,
    comprobanteFinal: null,
    notaCredito: null,
    prendas,
    pagos: [{ metodo: "efectivo", monto: adelanto, fecha: `${creadaEn}T16:20:00Z`, abono: false }],
    estante: estado === "abierta" ? `A-${String(n).padStart(2, "0")}` : null,
    canal: "tienda",
    modoEntrega: "recojo",
    direccionEnvio: null,
    enviadoEn: null,
    guiaEnvio: null,
    ...extra,
  };
};

export default async function PruebaApartados({ searchParams }: { searchParams: Promise<{ rol?: string; gestiona?: string; caja?: string; vacio?: string; marco?: string }> }) {
  const q = await searchParams;
  const lider = q.rol !== "integrante";
  const vacio = q.vacio === "1";
  const apartados = vacio ? [] : APARTADOS;
  const abiertos = apartados.filter((a) => a.estado === "abierta");
  const resumen = {
    porRecoger: abiertos.length,
    prendasGuardadas: abiertos.reduce((s, a) => s + a.prendas.reduce((t, p) => t + p.cantidad, 0), 0),
    enCustodia: abiertos.reduce((s, a) => s + a.adelanto, 0),
    enCustodiaEfectivo: abiertos.reduce((s, a) => s + a.pagos.filter((p) => p.metodo === "efectivo").reduce((t, p) => t + p.monto, 0), 0),
    porDevolver: apartados.filter((a) => a.estado === "liberada").length,
    montoPorDevolver: apartados.filter((a) => a.estado === "liberada").reduce((s, a) => s + a.adelanto, 0),
    vencenPronto: 2,
    vencidos: 1,
  };
  const panel = (
    <SedeActivaProveedor ubicacionId={SEDE} etiqueta="Trujillo" personaSesionId={null} esAdmin={lider} nombreSesion="Prueba">
      <ReglasDePrecioProveedor valor={{ reglas: REGLAS_DE_FABRICA, marcasConocidas: [], tejidosQueSeNotan: [] }}>
        <TrabajoPorSede key="apartados" ubicacionId={SEDE}>
          <ApartadosPanel
            ubicacionId={SEDE}
            ubicacionEtiqueta="Trujillo"
            abrir={null}
            hoy={HOY}
            cajaAbierta={q.caja !== "cerrada"}
            puedeGestionar={q.gestiona !== "0"}
            prendas={PRENDAS}
            filasMonto={[]}
            apartados={apartados}
            resumen={resumen}
            liberadosAhora={vacio ? 0 : 1}
            hayMas={false}
            avisos={{}}
            apagadas={[]}
            veClientas
            pedidos={vacio ? [] : PEDIDOS}
            tiendas={[{ id: "t-aqp", nombre: "AQP" }, { id: "t-lim", nombre: "LIM" }]}
            esLider={lider}
            lineasDesdeTicket={[]}
            redondeoEfectivo
            canales={[]}
          />
        </TrabajoPorSede>
      </ReglasDePrecioProveedor>
    </SedeActivaProveedor>
  );
  if (q.marco === "no") return <main className="p-6">{panel}</main>;
  return (
    <AppShell
      persona={{ nombre: "Prueba", personaId: null, rol: lider ? "lider" : "integrante", ubicacionId: SEDE, ubicacionEtiqueta: "Trujillo", ubicacionTipo: "tienda", puedeCambiarUbicacion: false, modulos: [...CLAVES_MODULO] }}
      ubicaciones={[{ id: SEDE, nombre: "Trujillo" }]}
      trasladosPorAtender={0}
    >
      {panel}
    </AppShell>
  );
}
